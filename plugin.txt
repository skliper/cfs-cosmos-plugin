<% (1..$cfs_fleet_globals_num_cfs_targets).each do |target_index| %>

    # Select the appropriate IP config input variable (declared above) for this cpu number
    <% if target_index == 1 %>
        <% target_name = target_1_name %>
        <% target_cmd_port = target_1_cmd_port %>
        <% target_tlm_port = target_1_tlm_port %>
    <% elsif target_index == 2 %>
        <% target_name = target_2_name %>
        <% target_cmd_port = target_2_cmd_port %>
        <% target_tlm_port = target_2_tlm_port %>
    <% elsif target_index == 3 %>
        <% target_name = target_3_name %>
        <% target_cmd_port = target_3_cmd_port %>
        <% target_tlm_port = target_3_tlm_port %>
    <% elsif target_index == 4 %>
        <% target_name = target_4_name %>
        <% target_cmd_port = target_4_cmd_port %>
        <% target_tlm_port = target_4_tlm_port %>
    <% else %>
        # This line should cause a plugin compilation error
        # If this line is hit, the loop count doesn't match the supported number of interfaces
        <% current_cfs_intf_ip = error %>
    <% end %>

    # Declare target
    TARGET CFS <%= target_name %>

    # Set up the FSW cmd/tlm interface
    INTERFACE <%= target_name %>_INTF udp_interface.rb <%= target_name %> <%= target_cmd_port %> <%= target_tlm_port %>
        MAP_TARGET <%= target_name %>
        PROTOCOL WRITE ProtocolCcsdsChecksum
        PROTOCOL WRITE ProtocolCcsdsLength
        PROTOCOL READ LengthProtocol 32 16 7 1 BIG_ENDIAN 0 nil
        PROTOCOL READ ProtocolEvsLog
<% end %>
